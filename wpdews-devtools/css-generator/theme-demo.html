Програмне управління темою

// Отримати поточну тему
const theme = window.themeToggle.get();
console.log(theme); // 'dark' або 'light'

// Встановити тему
window.themeToggle.set('light');
window.themeToggle.set('dark');

// Перемкнути тему
window.themeToggle.toggle();

// Використання в CSS
.my-element {
    background: var(--bg-primary);
    color: var(--text-primary);
    border: 2px solid var(--border-primary);
}
